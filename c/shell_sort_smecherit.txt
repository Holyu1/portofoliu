#define _CRT_SECURE_NO_WARNINGS
#include<stdio.h>
void sort_shell(double v[], int l)
{
    int i, j, inc;
    double a;
    for (inc = l / 2; inc > 0; inc = inc / 2)
        for (i = 0; i <= l-1; i++)
            if(i+inc<=l-1)
                if(v[i]>v[i+inc])
                {
                    a = v[i];
                    v[i] = v[i+ inc];
                    v[i + inc] = a;
                }
}


void main()
{
    int n, i;  double v[1000];
    printf("n="); scanf("%i", &n);
    for (i = 0; i < n; i++)
    {
        printf("v[%d]=", i);
        scanf("%lf", &v[i]);
    }
    sort_shell(v, n);
    printf("Vectorul sortat\n");
    for (i = 0; i < n; i++) printf("%8.4lf ", v[i]);
}
